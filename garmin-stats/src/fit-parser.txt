// ── FIT binary parser ─────────────────────────────────────────────────────
// Decodes Garmin .FIT files into typed activity + track-point records.

export interface FitActivity {
  filename: string;
  activity_date: string;   // ISO 8601
  date_only: string;       // YYYY-MM-DD
  sport: string;
  duration_sec: number | null;
  distance_m: number | null;
  avg_pace_minkm: number | null;
  calories: number | null;
  avg_hr: number | null;
  max_hr: number | null;
  avg_cadence: number | null;
  ascent_m: number | null;
  descent_m: number | null;
  avg_speed_ms: number | null;
  max_speed_ms: number | null;
}

export interface FitTrackPoint {
  elapsed_sec: number | null;
  distance_m: number | null;
  heart_rate: number | null;
  speed_ms: number | null;
  cadence: number | null;
  altitude_m: number | null;
  temperature: number | null;
  power: number | null;
  lat: number | null;
  lon: number | null;
}

export interface ParsedFit {
  activity: FitActivity;
  trackPoints: FitTrackPoint[];
}

// FIT timestamp epoch: 1989-12-31T00:00:00Z
const FIT_EPOCH_S = Math.floor(new Date("1989-12-31T00:00:00Z").getTime() / 1000);

const SPORT_MAP: Record<number, string> = {
  1: "running", 2: "cycling", 3: "transition", 4: "fitness_equipment",
  5: "swimming", 11: "walking", 15: "hiking", 17: "inline_skating",
};

// ── base type read descriptors ─────────────────────────────────────────────
interface BaseType {
  size: number;
  read: (buf: Buffer, pos: number, le: boolean) => number;
}

const BASE_TYPES: Record<number, BaseType> = {
  0:  { size:1, read:(b,p)=>b.readUInt8(p)   },
  1:  { size:1, read:(b,p)=>b.readInt8(p)    },
  2:  { size:1, read:(b,p)=>b.readUInt8(p)   },
  3:  { size:2, read:(b,p,le)=>le?b.readInt16LE(p):b.readInt16BE(p)   },
  4:  { size:2, read:(b,p,le)=>le?b.readUInt16LE(p):b.readUInt16BE(p) },
  5:  { size:4, read:(b,p,le)=>le?b.readInt32LE(p):b.readInt32BE(p)   },
  6:  { size:4, read:(b,p,le)=>le?b.readUInt32LE(p):b.readUInt32BE(p) },
  7:  { size:1, read:(b,p)=>b.readUInt8(p)   },  // string byte
  8:  { size:4, read:(b,p,le)=>le?b.readFloatLE(p):b.readFloatBE(p)   },
  9:  { size:8, read:(b,p,le)=>le?b.readDoubleLE(p):b.readDoubleBE(p) },
  10: { size:1, read:(b,p)=>b.readUInt8(p)   },
  11: { size:2, read:(b,p,le)=>le?b.readUInt16LE(p):b.readUInt16BE(p) },
  12: { size:4, read:(b,p,le)=>le?b.readUInt32LE(p):b.readUInt32BE(p) },
  13: { size:1, read:(b,p)=>b.readUInt8(p)   },
};

// global message numbers
const GMSG_FILE_ID = 0;
const GMSG_SESSION = 18;
const GMSG_RECORD  = 20;

// field numbers → property names for each global message
const FILE_ID_FIELDS: Record<number, string> = { 4:"time_created" };
const SESSION_FIELDS: Record<number, string> = {
  2:"sport", 7:"total_elapsed_time", 8:"total_timer_time",
  9:"total_distance", 11:"total_calories",
  14:"avg_speed", 15:"max_speed",
  16:"avg_heart_rate", 17:"max_heart_rate",
  24:"total_ascent", 25:"total_descent",
  56:"avg_cadence",
};
const RECORD_FIELDS: Record<number, string> = {
  0:"position_lat", 1:"position_long",
  2:"altitude", 3:"heart_rate", 4:"cadence",
  5:"distance", 6:"speed", 7:"power",
  13:"temperature", 29:"elapsed_time",
  78:"enhanced_altitude", 82:"enhanced_speed",
};

interface FieldDef { num: number; size: number; btc: number; }
interface MsgDef   { gNum: number; fields: FieldDef[]; le: boolean; }
type RawRecord     = Record<string, number | string>;

function parseFitBuffer(buf: Buffer): RawRecord[] {
  const headerLen = buf[0];
  if (headerLen < 12) throw new Error("Invalid FIT header");

  let pos = headerLen;
  const end = buf.length - 2;
  const defs: Record<number, MsgDef> = {};
  const records: RawRecord[] = [];

  while (pos < end) {
    const header = buf[pos];

    if (header & 0x80) { pos++; continue; }  // compressed timestamp

    const isDef  = !!(header & 0x40);
    const local  = header & 0x0f;

    if (isDef) {
      pos++;            // header
      pos++;            // reserved
      const arch  = buf[pos++];
      const le    = arch === 0;
      const gNum  = le ? buf.readUInt16LE(pos) : buf.readUInt16BE(pos);
      pos += 2;
      const nf    = buf[pos++];
      const fields: FieldDef[] = [];
      for (let f = 0; f < nf; f++) {
        const num  = buf[pos++];
        const size = buf[pos++];
        const btc  = buf[pos++] & 0x9f;
        fields.push({ num, size, btc });
      }
      defs[local] = { gNum, fields, le };

    } else {
      pos++;
      const def = defs[local];
      if (!def) { pos++; continue; }

      const row: RawRecord = { _g: def.gNum };

      for (const f of def.fields) {
        const bt = BASE_TYPES[f.btc] ?? BASE_TYPES[2];
        let val: number | string | null = null;

        if (f.btc === 7) {
          val = buf.slice(pos, pos + f.size).toString("latin1").replace(/\0/g, "");
        } else if (bt.size <= f.size && pos + f.size <= buf.length) {
          try { val = bt.read(buf, pos, def.le); } catch { /* skip */ }
        }
        pos += f.size;

        if (val === null) continue;

        if (def.gNum === GMSG_FILE_ID && FILE_ID_FIELDS[f.num])
          row[FILE_ID_FIELDS[f.num]] = val;
        if (def.gNum === GMSG_SESSION && SESSION_FIELDS[f.num])
          row[SESSION_FIELDS[f.num]] = val;
        if (def.gNum === GMSG_RECORD  && RECORD_FIELDS[f.num])
          row[RECORD_FIELDS[f.num]]  = val;
      }

      if ([GMSG_FILE_ID, GMSG_SESSION, GMSG_RECORD].includes(def.gNum))
        records.push(row);
    }
  }

  return records;
}

function resolveDate(raw: RawRecord[], filename: string): string {
  const fileId = raw.find(r => r._g === GMSG_FILE_ID);
  const tc = fileId?.time_created;

  if (typeof tc === "number" && tc > 0 && tc < 0xFFFFFFFF) {
    return new Date((tc + FIT_EPOCH_S) * 1000).toISOString().replace("Z", "");
  }

  // fallback: parse filename like "2024-08-01-07-30-00.fit"
  const m = filename.match(/(\d{4}-\d{2}-\d{2})-(\d{2})-(\d{2})-(\d{2})/);
  if (m) return `${m[1]}T${m[2]}:${m[3]}:${m[4]}`;

  return new Date().toISOString().replace("Z", "");
}

export function parseFit(buf: Buffer, filename: string): ParsedFit {
  const raw     = parseFitBuffer(buf);
  const session = raw.find(r => r._g === GMSG_SESSION) ?? {};
  const pts     = raw.filter(r => r._g === GMSG_RECORD);

  const actDate = resolveDate(raw, filename);
  const dateOnly = actDate.slice(0, 10);

  const sportCode = typeof session.sport === "number" ? session.sport : 0;
  const sport     = SPORT_MAP[sportCode] ?? "other";

  const durSec  = typeof session.total_elapsed_time === "number"
    ? session.total_elapsed_time / 1000 : null;
  const distM   = typeof session.total_distance === "number"
    ? session.total_distance / 100 : null;
  const avgSpd  = typeof session.avg_speed === "number"
    ? session.avg_speed / 1000 : null;
  const maxSpd  = typeof session.max_speed === "number"
    ? session.max_speed / 1000 : null;
  const pace    = avgSpd && avgSpd > 0 ? 1000 / avgSpd / 60 : null;

  const activity: FitActivity = {
    filename,
    activity_date:  actDate,
    date_only:      dateOnly,
    sport,
    duration_sec:   durSec,
    distance_m:     distM || null,
    avg_pace_minkm: pace,
    calories:       typeof session.total_calories  === "number" ? session.total_calories  : null,
    avg_hr:         typeof session.avg_heart_rate  === "number" ? session.avg_heart_rate  : null,
    max_hr:         typeof session.max_heart_rate  === "number" ? session.max_heart_rate  : null,
    avg_cadence:    typeof session.avg_cadence     === "number" ? session.avg_cadence     : null,
    ascent_m:       typeof session.total_ascent    === "number" ? session.total_ascent    : null,
    descent_m:      typeof session.total_descent   === "number" ? session.total_descent   : null,
    avg_speed_ms:   avgSpd,
    max_speed_ms:   maxSpd,
  };

  const trackPoints: FitTrackPoint[] = pts.map(r => {
    const spd = typeof r.enhanced_speed === "number" ? r.enhanced_speed / 1000
              : typeof r.speed          === "number" ? r.speed / 1000 : null;
    const alt = typeof r.enhanced_altitude === "number" ? r.enhanced_altitude / 5 - 500
              : typeof r.altitude          === "number" ? r.altitude / 5 - 500 : null;
    return {
      elapsed_sec: typeof r.elapsed_time   === "number" ? r.elapsed_time  : null,
      distance_m:  typeof r.distance       === "number" ? r.distance / 100 : null,
      heart_rate:  typeof r.heart_rate     === "number" ? r.heart_rate    : null,
      speed_ms:    spd,
      cadence:     typeof r.cadence        === "number" ? r.cadence       : null,
      altitude_m:  alt,
      temperature: typeof r.temperature    === "number" ? r.temperature   : null,
      power:       typeof r.power          === "number" ? r.power         : null,
      lat:         typeof r.position_lat   === "number" ? r.position_lat  / 11_930_465 : null,
      lon:         typeof r.position_long  === "number" ? r.position_long / 11_930_465 : null,
    };
  });

  return { activity, trackPoints };
}
